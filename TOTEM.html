// ============================================================ // SISTEMA DE PEDIDOS PARA RESTAURANTE (FAST-FOOD) // VERSÃO COM DUAS INTERFACES // ============================================================ // ==================== PÁGINA PRINCIPAL ==================== function doGet(e) { try { var pagina = e && e.parameter ? String(e.parameter.page || "").trim().toLowerCase() : ""; var api = e && e.parameter ? String(e.parameter.api || "").trim().toLowerCase() : ""; // Se for API, retorna JSON if (api) { return processarApi(e); } // Define qual página mostrar var arquivoHtml = ""; switch (pagina) { case "cliente": arquivoHtml = "cliente"; break; case "restaurante": arquivoHtml = "restaurante"; break; default: // Página inicial - escolha entre as interfaces arquivoHtml = "index"; } return HtmlService.createHtmlOutputFromFile(arquivoHtml) .setTitle(pagina === "cliente" ? "🍔 Faça seu Pedido" : "👨‍🍳 Painel do Restaurante") .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } catch (erro) { return ContentService.createTextOutput(JSON.stringify({ success: false, mensagem: "Erro: " + erro.toString() })).setMimeType(ContentService.MimeType.JSON); } } // ==================== PROCESSAR API ==================== function processarApi(e) { var params = e.parameter; var api = String(params.api || "").trim().toLowerCase(); var resultado; switch (api) { case "produtos": resultado = { success: true, data: getProdutos() }; break; case "pedidos": resultado = { success: true, data: getPedidosFila() }; break; case "pedido": var orderId = params.id || ""; if (!orderId) { resultado = { success: false, mensagem: "ID do pedido não informado." }; } else { var pedido = getPedidoById(orderId); resultado = { success: !!pedido, data: pedido, mensagem: pedido ? "" : "Pedido não encontrado." }; } break; case "proxima_comanda": resultado = { success: true, data: { comanda: getProximoNumeroComanda() } }; break; default: resultado = { success: false, mensagem: "API não encontrada." }; } return ContentService.createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } // ==================== POST ==================== function doPost(e) { try { var payload = {}; if (e && e.postData && e.postData.contents) { payload = JSON.parse(e.postData.contents); } var api = String(payload.api || "").trim().toLowerCase(); switch (api) { case "novopedido": return responderJson(criarPedido(payload.data)); case "atualizarstatus": return responderJson(atualizarStatusPedido(payload.data)); case "cancelarpedido": return responderJson(cancelarPedido(payload.data)); default: return responderJson({ success: false, mensagem: "API POST não encontrada." }); } } catch (erro) { return responderJson({ success: false, mensagem: "Erro: " + erro.toString() }); } } function responderJson(objeto) { return ContentService.createTextOutput(JSON.stringify(objeto)) .setMimeType(ContentService.MimeType.JSON); } // ==================== PEDIDOS ==================== function getPedidosFila() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName("PEDIDOS"); if (!sh) return []; var data = sh.getDataRange().getValues(); if (data.length <= 1) return []; var pedidos = []; var ordemStatus = { 'RECEBIDO': 0, 'PREPARANDO': 1, 'PRONTO': 2, 'ENTREGUE': 3, 'CANCELADO': 4 }; for (var i = data.length - 1; i >= 1; i--) { var row = data[i]; var status = String(row[8] || "RECEBIDO").trim().toUpperCase(); // Filtra cancelados para não aparecerem na fila principal if (status === 'CANCELADO') continue; pedidos.push({ id: String(row[0]).trim(), comanda: Number(row[11] || i), nome: String(row[1]).trim(), telefone: String(row[2]).trim(), itens: String(row[5]).trim(), quantidade: Number(row[6]) || 1, valor: Number(row[7]) || 0, status: status, statusLabel: getStatusLabel(status), data: String(row[10] || ""), observacao: String(row[12] || ""), tempoEspera: calcularTempoEspera(row[10]) }); } // Ordena por status (RECEBIDO primeiro) e depois por data pedidos.sort(function(a, b) { var orderA = ordemStatus[a.status] || 99; var orderB = ordemStatus[b.status] || 99; if (orderA !== orderB) return orderA - orderB; return new Date(a.data) - new Date(b.data); }); return pedidos; } catch (error) { Logger.log('Erro getPedidosFila: ' + error); return []; } } function getStatusLabel(status) { var labels = { 'RECEBIDO': '📥 Recebido', 'PREPARANDO': '👨‍🍳 Preparando', 'PRONTO': '✅ Pronto', 'ENTREGUE': '📦 Entregue', 'CANCELADO': '❌ Cancelado' }; return labels[status] || status; } function getStatusColor(status) { var cores = { 'RECEBIDO': '#ffc107', 'PREPARANDO': '#17a2b8', 'PRONTO': '#28a745', 'ENTREGUE': '#6c757d', 'CANCELADO': '#dc3545' }; return cores[status] || '#6c757d'; } function getProximoNumeroComanda() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName("PEDIDOS"); if (!sh) return 1; var data = sh.getDataRange().getValues(); if (data.length <= 1) return 1; var maxComanda = 0; for (var i = 1; i < data.length; i++) { var comanda = Number(data[i][11] || 0); if (comanda > maxComanda) maxComanda = comanda; } return maxComanda + 1; } catch (error) { return 1; } } function calcularTempoEspera(dataPedido) { if (!dataPedido) return "0 min"; try { var data = new Date(dataPedido); var agora = new Date(); var diffMs = agora - data; var diffMin = Math.floor(diffMs / 60000); if (diffMin < 1) return "Agora"; if (diffMin < 60) return diffMin + " min"; var horas = Math.floor(diffMin / 60); var minutos = diffMin % 60; return horas + "h " + minutos + "min"; } catch (e) { return "0 min"; } } function criarPedido(dados) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName("PEDIDOS"); if (!sh) { sh = ss.insertSheet("PEDIDOS"); sh.appendRow(["ID Pedido", "Nome", "Telefone", "E-mail", "Endereço", "Itens", "Quantidade", "Valor", "Status", "Pix Code", "Data", "Comanda", "Observação", "Itens Detalhados"]); } var comanda = getProximoNumeroComanda(); var orderId = "PED" + new Date().getTime(); var itensTexto = Array.isArray(dados.itens) ? dados.itens.join(", ") : dados.itens || ""; var itensDetalhados = Array.isArray(dados.itens) ? JSON.stringify(dados.itens) : dados.itens || ""; var valorTotal = Number(dados.valor) || 0; sh.appendRow([ orderId, dados.nome || "Cliente", dados.telefone || "", dados.email || "", dados.endereco || "Retirada no Balcão", itensTexto, dados.quantidade || 1, valorTotal, "RECEBIDO", "", new Date().toISOString(), comanda, dados.observacao || "", itensDetalhados ]); return { success: true, orderId: orderId, comanda: comanda, mensagem: "Pedido #" + comanda + " criado com sucesso!" }; } catch (error) { return { success: false, mensagem: error.toString() }; } } function atualizarStatusPedido(dados) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName("PEDIDOS"); if (!sh) return { success: false, mensagem: "Aba PEDIDOS não encontrada" }; var data = sh.getDataRange().getValues(); if (data.length <= 1) return { success: false, mensagem: "Nenhum pedido encontrado" }; var orderId = dados.id || dados.orderId; var novoStatus = String(dados.status || "").trim().toUpperCase(); var statusValidos = ['RECEBIDO', 'PREPARANDO', 'PRONTO', 'ENTREGUE', 'CANCELADO']; if (statusValidos.indexOf(novoStatus) === -1) { return { success: false, mensagem: "Status inválido: " + novoStatus }; } var linhaEncontrada = -1; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(orderId).trim()) { linhaEncontrada = i + 1; break; } } if (linhaEncontrada === -1) { return { success: false, mensagem: "Pedido não encontrado" }; } sh.getRange(linhaEncontrada, 9).setValue(novoStatus); return { success: true, mensagem: "Status atualizado para: " + getStatusLabel(novoStatus), status: novoStatus }; } catch (error) { return { success: false, mensagem: error.toString() }; } } function cancelarPedido(dados) { return atualizarStatusPedido({ id: dados.id, status: 'CANCELADO' }); } function getPedidoById(orderId) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName("PEDIDOS"); if (!sh) return null; var data = sh.getDataRange().getValues(); if (data.length <= 1) return null; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(orderId).trim()) { return { id: String(data[i][0]).trim(), comanda: Number(data[i][11] || i), nome: String(data[i][1]).trim(), telefone: String(data[i][2]).trim(), itens: String(data[i][5]).trim(), quantidade: Number(data[i][6]) || 1, valor: Number(data[i][7]) || 0, status: String(data[i][8] || "RECEBIDO").trim().toUpperCase(), data: String(data[i][10] || ""), observacao: String(data[i][12] || "") };
